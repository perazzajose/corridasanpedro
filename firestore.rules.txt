// Pegar en: Firebase Console → Firestore Database → pestaña "Reglas"
//
// Lectura pública (la galería la lee cualquier visitante sin login),
// escritura solo si estás logueado en /admin (Authentication).

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    match /galeria/{fotoId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /sponsors/{sponsorId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Bloquea todo lo demás por defecto.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
